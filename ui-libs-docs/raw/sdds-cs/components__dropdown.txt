Dropdown — multi-level dropdown menu, wraps a target via children (e.g. a Button). Only required prop: items. Item shape: { value: string|number, label: string, placement?, items?: ItemOption[] (nested), beforeList?, disabled?, contentLeft?, contentRight?, dividerBefore?, dividerAfter?, className?, trigger?, listMaxHeight?, hidden? }. Keyboard: matches W3C Combobox + partial TreeView (Tab closes, Enter/Space toggles/selects, Home/End/PageUp/PageDown/ArrowUp/Down navigate, ArrowRight opens submenu, ArrowLeft closes it).
LIVE EDITOR
function App() {
const items = [{ value: 'na', label: 'Северная Америка' }, { value: 'sa', label: 'Южная Америка', items: [{ value: 'br', label: 'Бразилия' }] }];
return <Dropdown items={items}><Button text="Список стран" /></Dropdown>;
}
