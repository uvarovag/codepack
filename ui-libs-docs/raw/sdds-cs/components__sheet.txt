Sheet — bottom sheet overlay. children = body content; contentHeader/contentFooter for pinned slots (isHeaderFixed/isFooterFixed keeps them fixed while body scrolls). withOverlay=true(default) dims background and clicking it closes; withOverlay=false lets user interact with content behind. withBlur adds blur to overlay. snapPoints (CSS sizes like '320px','50%','40dvh') for stop positions; initialSnapPoint; onSnapPointChange.
LIVE EDITOR
function App() {
const [opened, setOpened] = useState(false);
return (
<>
<Button onClick={() => setOpened(true)}>Открыть</Button>
<Sheet opened={opened} onClose={() => setOpened(false)} contentHeader={<h4>header</h4>} contentFooter={<p>footer</p>} isHeaderFixed isFooterFixed>
<div>body</div>
</Sheet>
</>
);
}
