Pagination. slots = max page-number buttons shown at once. count = total items (or pages if hasPerPage unset). type="default"|"compact" (compact shows only current page). hasPerPage shows a per-page-size dropdown (perPage/perPageList). hasQuickJump shows a "jump to page" input. leftContent/rightContent for custom nav buttons (first/prev/next/last). Controlled via value + perPage + onChange(page?, perPage?).
LIVE EDITOR
function App() {
const [page, setPage] = useState(1);
return <Pagination slots={9} count={2000} value={page} hasPerPage perPage={20} hasQuickJump onChange={(p, pp) => { if(p) setPage(p); }} />;
}
